cmake_minimum_required(VERSION 3.20)
project(program_runtime C CXX ASM)
get_filename_component(WORKSPACE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(CMAKE_C_STANDARD 11)
option(PROGRAM_SANITIZERS "Enable host address/undefined sanitizers" ON)
if(PROGRAM_SANITIZERS)
  add_compile_options(-O1 -g -fno-omit-frame-pointer -fsanitize=address,undefined -fno-sanitize-recover=all)
  add_link_options(-fsanitize=address,undefined)
endif()
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_LIST_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory("${WORKSPACE}/firmware/managed_components/lvgl__lvgl" lvgl)
set(QJS "${WORKSPACE}/.tools/quickjs-probe/quickjs-ng/quickjs-ng")
add_library(quickjs STATIC "${QJS}/quickjs.c" "${QJS}/dtoa.c" "${QJS}/libunicode.c" "${QJS}/libregexp.c")
target_compile_definitions(quickjs PRIVATE _GNU_SOURCE)
target_compile_options(quickjs PRIVATE -funsigned-char)
target_include_directories(quickjs SYSTEM PUBLIC "${QJS}")
set(IDF "${WORKSPACE}/.tools/esp-idf-v5.5.5")
set(MBEDTLS "${IDF}/components/mbedtls/mbedtls")
add_library(program_crypto STATIC "${MBEDTLS}/library/sha256.c" "${MBEDTLS}/library/base64.c" "${MBEDTLS}/library/platform_util.c")
target_include_directories(program_crypto SYSTEM PUBLIC "${MBEDTLS}/include")
target_include_directories(program_crypto PRIVATE "${CMAKE_CURRENT_LIST_DIR}")
target_compile_definitions(program_crypto PUBLIC MBEDTLS_CONFIG_FILE="mbedtls_config.h")
add_library(program_json STATIC "${IDF}/components/json/cJSON/cJSON.c")
target_include_directories(program_json SYSTEM PUBLIC "${IDF}/components/json/cJSON")
add_library(program_core STATIC program_runtime.c program_package.c)
target_compile_options(program_core PRIVATE -Wall -Wextra -Werror -Wno-misleading-indentation)
target_include_directories(program_core PUBLIC "${CMAKE_CURRENT_LIST_DIR}")
target_link_libraries(program_core PUBLIC quickjs lvgl program_json program_crypto m)
add_executable(program_tests program_tests.c)
target_compile_options(program_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
target_link_libraries(program_tests PRIVATE program_core)
add_executable(startup_tests startup_tests.c)
target_compile_options(startup_tests PRIVATE -Wall -Wextra -Werror -Wno-misleading-indentation -UNDEBUG)
target_link_libraries(startup_tests PRIVATE program_core)
add_executable(allocator_tests allocator_tests.c)
target_include_directories(allocator_tests PRIVATE "${CMAKE_CURRENT_LIST_DIR}/test-esp")
target_compile_options(allocator_tests PRIVATE -Wall -Wextra -Werror -Wno-misleading-indentation -UNDEBUG)
target_link_libraries(allocator_tests PRIVATE program_core)
add_executable(program_runner program_runner.c)
target_compile_options(program_runner PRIVATE -Wall -Wextra -Werror)
target_link_libraries(program_runner PRIVATE program_core)
