cmake_minimum_required(VERSION 3.20)
project(programmable_ui_probe C CXX ASM)
get_filename_component(WORKSPACE "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -O1 -g -fno-omit-frame-pointer -fsanitize=address,undefined -fno-sanitize-recover=all")
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -O1 -g -fno-omit-frame-pointer -fsanitize=address,undefined -fno-sanitize-recover=all")
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_LIST_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory("${WORKSPACE}/firmware/managed_components/lvgl__lvgl" lvgl)
set(QJS "${WORKSPACE}/.tools/quickjs-probe/quickjs-ng/quickjs-ng")
add_library(quickjs STATIC "${QJS}/quickjs.c" "${QJS}/dtoa.c" "${QJS}/libunicode.c" "${QJS}/libregexp.c")
target_compile_definitions(quickjs PRIVATE _GNU_SOURCE)
target_compile_options(quickjs PRIVATE -funsigned-char)
target_include_directories(quickjs SYSTEM PUBLIC "${QJS}")
add_executable(ui_probe ui_probe.c)
target_compile_options(ui_probe PRIVATE -Wall -Wextra -Werror -UNDEBUG)
target_link_libraries(ui_probe PRIVATE quickjs lvgl m pthread)
